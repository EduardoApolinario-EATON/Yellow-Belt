Sub Novo_zip_vazio(sPath)
 'Se já existir um arquivo zip, apagá-lo e criar um novo
 If Len(Dir(sPath)) > 0 Then Kill sPath
 Open sPath For Output As #1
 Print #1, Chr$(80) & Chr$(75) & Chr$(5) & Chr$(6) & String(18, 0)
 Close #1
End Sub
